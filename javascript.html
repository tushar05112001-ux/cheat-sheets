JavaScript Cheat Sheet

Variables


let name = "Tushar";
const age = 24;
var city = "Delhi";

Data Types


String
Number
Boolean
Undefined
Null
Object
Array
Symbol
BigInt

Input


const value = document.querySelector("#input").value;

Conditions


if (age >= 18) {
    console.log("Adult");
} else {
    console.log("Minor");
}

Loops


for (let i = 0; i < 5; i++) {
    console.log(i);
}

while (condition) {
    // code
}

Functions


function add(a, b) {
    return a + b;
}

const multiply = (a, b) => {
    return a * b;
};

Arrays


const fruits = ["Apple", "Banana", "Mango"];

fruits.push("Orange");
fruits.pop();

fruits.map(item => console.log(item));

fruits.filter(item => item !== "Apple");

Objects


const user = {
    name: "Tushar",
    age: 24
};

console.log(user.name);

DOM


const element = document.querySelector("#title");

element.textContent = "Hello";

element.classList.add("active");

element.addEventListener("click", () => {
    console.log("Clicked");
});

Fetch API


fetch("https://example.com/api")
    .then(response => response.json())
    .then(data => {
        console.log(data);
    })
    .catch(error => {
        console.log(error);
    });
