React Cheat Sheet

Create React App


npm create vite@latest my-app
cd my-app
npm install
npm run dev

Component


function App() {
    return (
        <h1>Hello React</h1>
    );
}

export default App;

JSX


const name = "Tushar";

return (
    <h1>Hello {name}</h1>
);

Props


function User({ name }) {
    return <h1>{name}</h1>;
}

<User name="Tushar" />

useState


import { useState } from "react";

function Counter() {
    const [count, setCount] = useState(0);

    return (
        <button onClick={() => setCount(count + 1)}>
            {count}
        </button>
    );
}

useEffect


import { useEffect } from "react";

useEffect(() => {
    console.log("Component loaded");
}, []);

Array Rendering


const users = ["John", "Alex", "Sam"];

return (
    <div>
        {users.map((user, index) => (
            <p key={index}>{user}</p>
        ))}
    </div>
);

Event Handling


function handleClick() {
    console.log("Clicked");
}

<button onClick={handleClick}>
    Click Me
</button>
